#======================================== 库源文件列表 ================================================================
# 显式列出源文件，替代 aux_source_directory() 以追踪文件变化
set(LIB_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw/neon_osd.c
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw/osd_base.c
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw/cpu_osd.c
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw/frontRes.c
)

#======================================== 库的生成 =================================================================
# 静态库
add_library(neonOsd_static STATIC ${LIB_SOURCES})
target_include_directories(neonOsd_static PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw
)
target_compile_options(neonOsd_static PRIVATE
    -Wall
    -Wno-unused
    -Wconversion
    -Wno-uninitialized
    -Wno-sign-conversion
    -mfloat-abi=softfp
    -mfpu=neon-vfpv4
)
set_target_properties(neonOsd_static PROPERTIES
    OUTPUT_NAME neonOsd
    COMPILE_FLAGS " -O2 "
)

# 共享库
add_library(neonOsd_shared SHARED ${LIB_SOURCES})
target_include_directories(neonOsd_shared PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw
)
target_compile_options(neonOsd_shared PRIVATE
    -Wall
    -Wno-unused
    -Wconversion
    -Wno-uninitialized
    -Wno-sign-conversion
    -mfloat-abi=softfp
    -mfpu=neon-vfpv4
    -fPIE
    -pie
    -fPIC
)
set_target_properties(neonOsd_shared PROPERTIES
    OUTPUT_NAME neonOsd
    COMPILE_FLAGS " -O0 "
    VERSION 1.2.3
    SOVERSION 1
)

# 安装共享库和头文件:
install(TARGETS neonOsd_shared neonOsd_static
    LIBRARY DESTINATION ${CMAKE_SOURCE_DIR}/install/lib
    ARCHIVE DESTINATION ${CMAKE_SOURCE_DIR}/install/lib
)
install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw/
    DESTINATION ${CMAKE_SOURCE_DIR}/install/include
    FILES_MATCHING PATTERN "*.h"
)

#========================================== 测试 demo 的生成 =======================================================
# demo 源文件
set(DEMO_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/demo/src/test.c
    ${CMAKE_CURRENT_SOURCE_DIR}/demo/src/osd_test.c
    ${CMAKE_CURRENT_SOURCE_DIR}/demo/src/neon_intrinsics_test.c
    ${CMAKE_CURRENT_SOURCE_DIR}/demo/src/build_time.c
)

add_executable(neonOsdDemo ${DEMO_SOURCES})

target_include_directories(neonOsdDemo PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/demo/inc
    ${CMAKE_CURRENT_SOURCE_DIR}/osd_Draw
)

# 链接静态库和系统库
target_link_libraries(neonOsdDemo PRIVATE neonOsd_static m dl)

# 配置依赖关系
add_dependencies(neonOsdDemo neonOsd_shared neonOsd_static)

# 复制 compile_commands.json 到项目根目录，方便 IDE 查找
add_custom_command(TARGET neonOsdDemo POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        ${PROJECT_BINARY_DIR}/compile_commands.json
        ${PROJECT_SOURCE_DIR}/compile_commands.json
    COMMENT "Copying compile_commands.json to project root"
)

# 安装 demo
install(TARGETS neonOsdDemo DESTINATION ${CMAKE_SOURCE_DIR}/install
    PERMISSIONS OWNER_EXECUTE OWNER_WRITE OWNER_READ
            GROUP_EXECUTE GROUP_READ WORLD_EXECUTE
)
